================================================================================
               z_slog Real-Time Colored Logger Implementation Guide
================================================================================

OVERVIEW
--------
The z_slog logging system provides:
- Colored terminal output with emoji icons
- Real-time file logging to .ansi files
- Per-user activity tracking with SESSION START/END markers
- Function entry/exit logging with parameter values
- Yellow millisecond highlighting in timestamps
- VSCode integrated terminal support
- macOS Terminal integration with dark mode
- Compliance checker tool for verifying logging coverage

================================================================================
                              QUICK START
================================================================================

1. Create the z_slog package directory:
   mkdir -p your-project/z_slog

2. Create three files:
   - z_slog/icons.go      (Color constants and emoji icons)
   - z_slog/logger.go     (Main logger with colorHandler)
   - z_slog/user_logger.go (Per-user session logging)

3. Initialize in main.go:
   z_slog.InitFromEnv()
   defer z_slog.Close()
   defer z_slog.CloseAllUserLogs()

import (
    "context"

    "strings"
    "time"

var (
    DEBUG   bool
    logFile *os.File
)

// colorHandler is a custom slog handler that preserves ANSI color codes
type colorHandler struct {
    writer io.Writer
    level  slog.Level
}

func (h *colorHandler) Enabled(_ context.Context, level slog.Level) bool {
    return level >= h.level
}

func (h *colorHandler) Handle(_ context.Context, r slog.Record) error {
    buf := make([]byte, 0, 256)

    // Timestamp with yellow milliseconds: t=02-30-45-123 PM
    buf = append(buf, "t="...)
    buf = r.Time.AppendFormat(buf, "03-04-05-")
    buf = append(buf, ColorYellow...)
    buf = append(buf, fmt.Sprintf("%03d", r.Time.Nanosecond()/1e6)...)
    buf = append(buf, ColorReset...)
    buf = append(buf, ' ')
    buf = r.Time.AppendFormat(buf, "PM")
    buf = append(buf, ' ')

    // Level with color
    buf = append(buf, "level="...)
    levelStr := ""
    switch r.Level {
    case slog.LevelDebug:
        levelStr = ColorCyan + IconDebug + " DEBUG" + ColorReset
    case slog.LevelInfo:
        levelStr = ColorBlue + IconInfo + "  INFO " + ColorReset
    case slog.LevelWarn:
        levelStr = ColorYellow + IconWarning + "  WARN " + ColorReset
    case slog.LevelError:
        levelStr = ColorRed + IconError + " ERROR" + ColorReset
    default:
        levelStr = r.Level.String()
    }
    buf = append(buf, levelStr...)
    buf = append(buf, ' ')

    // Message (preserve ANSI codes)
    buf = append(buf, "msg="...)
    buf = append(buf, r.Message...)

    // Attributes
    r.Attrs(func(a slog.Attr) bool {
        buf = append(buf, ' ')
        buf = append(buf, a.Key...)
        buf = append(buf, '=')
        buf = append(buf, fmt.Sprint(a.Value.Any())...)
        return true
    })

    buf = append(buf, '\n')
    _, err := h.writer.Write(buf)

    if logFile != nil {
        logFile.Sync()
    }

    return err
}

func (h *colorHandler) WithAttrs(attrs []slog.Attr) slog.Handler { return h }
func (h *colorHandler) WithGroup(name string) slog.Handler      { return h }

// Init initializes the logger with file output
func Init(debug bool) {
    DEBUG = debug

    if logFile != nil {
        logFile.Close()
    }

    // Create logs directory inside z_slog folder
    logsDir := "z_slog/logs"
    if err := os.MkdirAll(logsDir, 0755); err != nil {
        fmt.Fprintf(os.Stderr, "Failed to create logs directory: %v\n", err)
    }

    // Timestamped log file: output-YYYY-MM-DD-HH-MM.ansi
    timestamp := time.Now().Format("2006-01-02-15-04")
    logFileName := fmt.Sprintf("output-%s.ansi", timestamp)
    logFilePath := filepath.Join(logsDir, logFileName)

    var err error
    logFile, err = os.Create(logFilePath)
    if err != nil {
        fmt.Fprintf(os.Stderr, "Failed to create log file: %v\n", err)
        logFile = nil
    }

    level := slog.LevelInfo
    if debug {
        level = slog.LevelDebug
    }

    var writer io.Writer
    if logFile != nil {
        writer = io.MultiWriter(os.Stdout, logFile)
        fmt.Printf("Logging to: %s\n", logFilePath)
    } else {
        writer = os.Stdout
    }

    handler := &colorHandler{writer: writer, level: level}
    logger := slog.New(handler)
    slog.SetDefault(logger)
}

func InitFromEnv() {
    debugMode := strings.ToLower(os.Getenv("DEBUG_MODE")) == "true"
    Init(debugMode)
}

func Close() {
    if logFile != nil {
        logFile.Close()
        logFile = nil
    }
}

// FunctionEntry logs function entry with cyan color and CODE icon (not INFO icon!)
func FunctionEntry(functionName string) {
    slog.Info(ColorCyan + IconCode + " Function " + IconArrowR + " " + functionName + ColorReset)
}

// Values logs function parameters (cyan names, bright magenta values)
func Values(params map[string]any) {
    if len(params) == 0 {
        return
    }

    var parts []string
    for key, value := range params {
        var valueStr string
        switch v := value.(type) {
        case string:
            valueStr = fmt.Sprintf("\"%s\"", v)
        case nil:
            valueStr = "null"
        case int, int8, int16, int32, int64, uint, uint8, uint16, uint32, uint64, float32, float64, bool:
            valueStr = fmt.Sprintf("%v", v)
        default:
            jsonBytes, err := json.MarshalIndent(v, "", "  ")
            if err != nil {
                valueStr = fmt.Sprintf("%v", v)
            } else {
                valueStr = string(jsonBytes)
            }
        }
        parts = append(parts, fmt.Sprintf("%s%s%s: %s%s%s",
            ColorCyan, key, ColorReset,
            ColorBrightMagenta, valueStr, ColorReset))
    }

    paramsStr := strings.Join(parts, "\n")
    msg := fmt.Sprintf("%s%s VALUES%s\n%s", ColorMagenta, IconValues, ColorReset, paramsStr)
    slog.Info(msg)
}

// ReturnValues logs function return values (purple names, bright green values)
func ReturnValues(returns map[string]any) {
    if len(returns) == 0 {
        return
    }

    var parts []string
    for key, value := range returns {
        var valueStr string
        switch v := value.(type) {
        case string:
            valueStr = fmt.Sprintf("\"%s\"", v)
        case nil:
            valueStr = "null"
        case int, int8, int16, int32, int64, uint, uint8, uint16, uint32, uint64, float32, float64, bool:
            valueStr = fmt.Sprintf("%v", v)
        default:
            jsonBytes, err := json.MarshalIndent(v, "", "  ")
            if err != nil {
                valueStr = fmt.Sprintf("%v", v)
            } else {
                valueStr = string(jsonBytes)
            }
        }
        parts = append(parts, fmt.Sprintf("%s%s%s: %s%s%s",
            ColorPurple, key, ColorReset,
            ColorBrightGreen, valueStr, ColorReset))
    }

    returnsStr := strings.Join(parts, "\n")
    msg := fmt.Sprintf("%s%s RETURNED%s\n%s", ColorGreen, IconReturn, ColorReset, returnsStr)
    slog.Info(msg)
}

package z_slog

import (
    "fmt"
    "os"
    "path/filepath"
    "sync"
    "time"
)

type UserLogManager struct {
    mu        sync.RWMutex
    userFiles map[string]*os.File
}

var userLogManager = &UserLogManager{
    userFiles: make(map[string]*os.File),
}

// StartUserLogging opens a log file for the user on login
func StartUserLogging(username string) error {
    FunctionEntry("StartUserLogging")
    Values(map[string]any{"username": username})

    userLogManager.mu.Lock()
    defer userLogManager.mu.Unlock()

    // Create user logs directory inside z_slog/logs
    userLogsDir := filepath.Join("z_slog", "logs", "users")
    if err := os.MkdirAll(userLogsDir, 0755); err != nil {
        return fmt.Errorf("failed to create user logs directory: %w", err)
    }

    // Filename: username-YYYY-MM-DD.ansi
    date := time.Now().Format("2006-01-02")
    filename := fmt.Sprintf("%s-%s.ansi", username, date)
    filePath := filepath.Join(userLogsDir, filename)

    file, err := os.OpenFile(filePath, os.O_CREATE|os.O_APPEND|os.O_WRONLY, 0644)
    if err != nil {
        return fmt.Errorf("failed to open user log file: %w", err)
    }

    if oldFile, exists := userLogManager.userFiles[username]; exists {
        oldFile.Close()
    }

    userLogManager.userFiles[username] = file

    // Green SESSION START marker
    sessionStart := fmt.Sprintf("\n%s=== SESSION START ===%s\n", ColorGreen, ColorReset)
    file.WriteString(sessionStart)
    file.Sync()

    return nil
}

// LogUserAction logs a message to the user's log file
func LogUserAction(username, msg string) {
    if username == "" {
        return
    }

    userLogManager.mu.RLock()
    file, exists := userLogManager.userFiles[username]
    userLogManager.mu.RUnlock()

    if !exists {
        return
    }

    // Timestamp with yellow milliseconds
    now := time.Now()
    timestampBase := now.Format("03-04-05-")
    milliseconds := fmt.Sprintf("%03d", now.Nanosecond()/1e6)  // CORRECT!
    ampm := now.Format("PM")

    timestamp := fmt.Sprintf("t=%s%s%s%s %s", timestampBase, ColorYellow, milliseconds, ColorReset, ampm)
    logLine := fmt.Sprintf("%s %s\n", timestamp, msg)
    file.WriteString(logLine)
    file.Sync()
}

// StopUserLogging closes the user's log file on logout
func StopUserLogging(username string) {
    FunctionEntry("StopUserLogging")
    Values(map[string]any{"username": username})

    userLogManager.mu.Lock()
    defer userLogManager.mu.Unlock()

    if file, exists := userLogManager.userFiles[username]; exists {
        // Red SESSION END marker
        sessionEnd := fmt.Sprintf("%s=== SESSION END ===%s\n\n", ColorRed, ColorReset)
        file.WriteString(sessionEnd)
        file.Sync()
        file.Close()
        delete(userLogManager.userFiles, username)
    }
}

// CloseAllUserLogs closes all user logs on server shutdown
func CloseAllUserLogs() {
    userLogManager.mu.Lock()
    defer userLogManager.mu.Unlock()

    for username, file := range userLogManager.userFiles {
        sessionEnd := fmt.Sprintf("%s=== SESSION END ===%s\n\n", ColorRed, ColorReset)
        file.WriteString(sessionEnd)
        file.Sync()
        file.Close()
        delete(userLogManager.userFiles, username)
    }
}

package api

import (
    "fmt"
    "net/http"
    "your-project/z_slog"
)

// LogUserAction logs an action to the user's specific log file
func LogUserAction(r *http.Request, functionName string) {
    user := GetUserFromContext(r)
    if user == nil {
        return
    }
    msg := fmt.Sprintf("%s%s Function %s %s%s",
        z_slog.ColorCyan, z_slog.IconCode, z_slog.IconArrowR, functionName, z_slog.ColorReset)
    z_slog.LogUserAction(user.Name, msg)
}

// LogUserActionWithDetails logs an action with additional details
func LogUserActionWithDetails(r *http.Request, functionName string, details string) {
    user := GetUserFromContext(r)
    if user == nil {
        return
    }
    msg := fmt.Sprintf("%s%s Function %s %s | %s%s",
        z_slog.ColorCyan, z_slog.IconCode, z_slog.IconArrowR, functionName, details, z_slog.ColorReset)
    z_slog.LogUserAction(user.Name, msg)
}

================================================================================
                   COMPLIANCE CHECKER TOOL (IMPORTANT!)
================================================================================

The compliance checker verifies that all Go functions have proper z_slog logging.
It scans all .go files and reports which functions are missing FunctionEntry and
ReturnValues calls.

FILE: z_slog/runners/update_logger.go
-------------------------------------

Run with: go run z_slog/runners/update_logger.go

Or add to Makefile:
    compliance:
        @go run z_slog/runners/update_logger.go

The checker scans three sections:
- BACKEND: internal/** directories
- FRONTEND: frontend/** directories (shows "n/a" - Go logger is backend only)
- ROOT: cmd/server and z_slog directories

--------------------------------------------------------------------------------
                    INFRASTRUCTURE FUNCTIONS (EXCLUDED)
--------------------------------------------------------------------------------

These functions are intentionally excluded from compliance checking because they
are infrastructure/helper code that should NOT have FunctionEntry/ReturnValues:

LOGGER INFRASTRUCTURE (z_slog package):
    - Enabled, Handle, WithAttrs, WithGroup (colorHandler methods)
    - Init, InitFromEnv, Close (logger lifecycle)
    - Debug, Info, Warn, Error, etc. (semantic logging functions)
    - FunctionEntry, Values, ReturnValues (the logging functions themselves!)
    - Auth, Video, Token, User, Database, API, Upload, Download
    - Success, Failed, Warning, Server, Network, Security
    - Critical, Start, Stop, Config, Report
    - StartUserLogging, LogUserAction, StopUserLogging, CloseAllUserLogs

API INFRASTRUCTURE (internal/api package):
    - respondWithJSON, respondWithError (response helpers)
    - LogUserAction, LogUserActionWithDetails (thin wrappers)

HTTP MIDDLEWARE INTERNALS:
    - WriteHeader (responseWrapper method)

ANONYMOUS FUNCTIONS:
    - http.HandlerFunc closures inside middleware
    - Callback functions passed to other functions

Total: ~45 infrastructure functions excluded from compliance checks.

package main

import (
    "fmt"
    "log"
    "net/http"
    "os"

    "github.com/gorilla/mux"
    "github.com/joho/godotenv"
    "your-project/internal/api"
    "your-project/z_slog"
)

func main() {
    // Initialize z_slog logger FIRST
    z_slog.InitFromEnv()
    defer z_slog.Close()
    defer z_slog.CloseAllUserLogs()

    z_slog.FunctionEntry("main")

    if err := godotenv.Load(); err != nil {
        log.Println("Warning: .env file not found")
    }

    serverPort := getEnv("SERVER_PORT", "8080")

    z_slog.Start("Application starting")
    z_slog.Config("Configuration loaded", "port", serverPort)

    router := mux.NewRouter()

    addr := fmt.Sprintf(":%s", serverPort)
    z_slog.Server(fmt.Sprintf("Server listening on http://localhost%s", addr))

    if err := http.ListenAndServe(addr, router); err != nil {
        z_slog.Critical(fmt.Sprintf("Server failed: %v", err))
    }
}

    less -R z_slog/logs/$(ls -t z_slog/logs/output-*.ansi | head -1)

View specific user's log:
    less -R z_slog/logs/users/Sam-2026-01-29.ansi

Tail logs in real-time:
    tail -f z_slog/logs/output-2026-01-29-14-30.ansi

Search logs:
    grep -i "error" z_slog/logs/output-*.ansi
    grep "Function -> CreateVideo" z_slog/logs/output-*.ansi

================================================================================
                      CRITICAL BUG FIX: MILLISECONDS
================================================================================

IMPORTANT: Go's time.Format("000") does NOT extract milliseconds!

WRONG - Always outputs "000":
    milliseconds := now.Format("000")

CORRECT - Extracts actual milliseconds (0-999):
    milliseconds := fmt.Sprintf("%03d", now.Nanosecond()/1e6)

================================================================================
                            TROUBLESHOOTING
================================================================================

+--------------------------------+---------------------------------------------+
| Issue                          | Solution                                    |
+--------------------------------+---------------------------------------------+
| Colors not showing in terminal | Use "less -R" instead of "less"             |
+--------------------------------+---------------------------------------------+
| Colors not showing in VSCode   | Add "console": "integratedTerminal"         |
+--------------------------------+---------------------------------------------+
| Milliseconds showing "000"     | Use fmt.Sprintf("%03d", now.Nanosecond()/1e6)|
+--------------------------------+---------------------------------------------+
| User logs not created          | Ensure StartUserLogging() called on login   |
+--------------------------------+---------------------------------------------+
| SESSION END not appearing      | Ensure StopUserLogging() called on logout   |
+--------------------------------+---------------------------------------------+
| Quoted ANSI codes in output    | Use custom colorHandler, not slog.TextHandler|
+--------------------------------+---------------------------------------------+
| Logs not flushing in real-time | Ensure file.Sync() is called after writes   |
+--------------------------------+---------------------------------------------+
| Duplicate icons in output      | FunctionEntry uses code icon, not info icon |
+--------------------------------+---------------------------------------------+
| Too much noise in logs         | Don't log infrastructure functions          |
+--------------------------------+---------------------------------------------+
| Compliance checker shows errors| Add missing FunctionEntry/ReturnValues      |
+--------------------------------+---------------------------------------------+
| Function wrongly flagged       | Add to infrastructureFunctions exclusion map|
+--------------------------------+---------------------------------------------+
